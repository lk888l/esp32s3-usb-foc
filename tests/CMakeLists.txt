cmake_minimum_required(VERSION 3.20)
project(knob_host_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
add_executable(control_tests control_tests.cpp ../components/knob_core/src/control.cpp)
target_include_directories(control_tests PRIVATE ../components/knob_core/include)
target_compile_options(control_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME foc_and_haptics COMMAND control_tests)
add_executable(current_limit_tests current_limit_tests.cpp ../components/knob_core/src/control.cpp)
target_include_directories(current_limit_tests PRIVATE ../components/knob_core/include)
target_compile_options(current_limit_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME current_limit_transients COMMAND current_limit_tests)
add_executable(speed_limit_tests speed_limit_tests.cpp ../components/knob_core/src/control.cpp)
target_include_directories(speed_limit_tests PRIVATE ../components/knob_core/include)
target_compile_options(speed_limit_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME haptic_speed_transitions COMMAND speed_limit_tests)
